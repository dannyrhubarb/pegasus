# Crawl policy for pegasusmoonlander.com (tools/build-site.sh serves this
# file at the site root; a robots.txt is per host, so it says nothing about
# the API host or the replay CDN — those answer X-Robots-Tag: noindex from
# pegasus-backend instead). Everything not listed is crawlable.
#
# One group, no blank lines inside it: some parsers (Python's
# urllib.robotparser among them) end a group at a blank line and drop the
# rules after it, and some apply the first matching rule rather than the
# longest, so there is deliberately no "Allow: /" ahead of the rules.
User-agent: *
# PR previews: every pr-<n>/ is a full copy of the game pointed at the
# staging backend — duplicate content, never a search result. A plain
# prefix (rules are prefix matches) rather than /pr-*/, so it needs no
# wildcard support; nothing else at the root starts with "pr-".
Disallow: /pr-
# The sideload APK published by the Android release workflow (a binary
# download, not a page).
Disallow: /app/
# The level editor is deliberately unlinked from the game UI (see
# CLAUDE.md "Level editor & custom drafts").
Disallow: /play/editor.html
